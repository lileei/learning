# 用户态 Socket API 实践：在QEMU 环境中，用 vcan（虚拟 CAN 接口） 跑通 CAN 帧的收发
## 1）下载 can-utils
git clone https://github.com/linux-can/can-utils.git
cd can-utils/
## 2）使用交叉编译器静态编译 can-utils （一定要有 LDFLAGS="-static" 选项）
make CC=arm-linux-gnueabi-gcc CFLAGS="-static" LDFLAGS="-static"
## 3）编译完成后，将生成的工具复制到 BusyBox 的 _install/bin/ 目录
cp cansend candump cangen cansniffer ~/busybox/busybox-1.36.1/_install/bin/
## 4）重新打包 initrd 并启动 QEMU
## 5）进入 QEMU 终端后，执行以下命令创建虚拟 CAN 接口
    ### [1] 加载 vcan 模块（如果编译为模块）
    modprobe vcan
    ### [2] 创建 vcan0 接口
    ip link add dev vcan0 type vcan
    ### [3] 启用接口
    ip link set vcan0 up
    ### [4] 验证接口存在
    ip link show vcan0
    ### [5] 使用 candump 后台接收，输出到文件
    candump vcan0 > /tmp/can.log &
    ### [6] 使用 cansend 发送报文
    cansend vcan0 123#1122334455667788
    ### [7] 查看接收记录
    cat /tmp/can.log
    ### [8] 结束后杀掉后台 candump
    kill %1
 
# ftrace (Function Trace)
## 1）打开支持 ftrace 的内核选项
## 2）挂载 tracefs 目录
    ### <a> （永久生效）编辑 /etc/fstab 文件，添加以下内容
    debugfs /sys/kernel/debug debugfs defaults 0 0      // （debugfs 方式）
    或者
    tracefs /sys/kernel/tracing tracefs defaults 0 0    // （tracefs 方式，推荐）
    ### <b> （运行时挂载）
    mount -t debugfs nodev /sys/kernel/debug
    mount -t tracefs nodev /sys/kernel/tracing
## 3）查看可用的 tracers 列表（当前位于 /sys/kernel/debug/tracing 目录下）
cat available_tracers
## 4）设置当前使用的 tracers（以 function_graph 为例）
echo function_graph > current_tracer
## 5）指定要追踪的函数（以 do_sys_open 为例）
echo do_sys_open > set_ftrace_filter
## 6）开始追踪
echo 1 > tracing_on
## 7）执行你的操作（比如触发一次文件打开）
## 8）停止追踪
echo 0 > tracing_on
## 9）查看结果
less trace
